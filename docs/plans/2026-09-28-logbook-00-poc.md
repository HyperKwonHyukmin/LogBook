```python
# poc/drm_check.py
"""실험 1: 공유 폴더의 문서가 평문으로 읽히는지 확인한다.

사용법: python drm_check.py <파일 또는 폴더> [...]
출력: 파일별 {path, size, sniff, open} JSON 배열. DRM 파일이 하나라도 있으면 종료 코드 1.
"""
import json
import sys
from pathlib import Path

DRM_MAGIC = b"HHIDRMC"


def sniff(head: bytes) -> str:
    """파일 첫 바이트로 형식을 판별한다."""
    if head.startswith(DRM_MAGIC):
        return "drm"
    if head.startswith(b"PK\x03\x04"):
        return "zip"  # pptx·xlsx·docx 는 zip 컨테이너
    if head.startswith(b"%PDF"):
        return "pdf"
    return "unknown"


def try_open(path: Path) -> str:
    """실제 추출 라이브러리로 열어 본다. 성공 시 'ok ...', 실패 시 'fail ...'."""
    ext = path.suffix.lower()
    try:
        if ext == ".pdf":
            import fitz

            with fitz.open(path) as doc:
                return f"ok pages={doc.page_count}"
        if ext == ".pptx":
            from pptx import Presentation

            return f"ok slides={len(Presentation(str(path)).slides)}"
        if ext == ".xlsx":
            from openpyxl import load_workbook

            wb = load_workbook(path, read_only=True)
            try:
                return f"ok sheets={len(wb.sheetnames)}"
            finally:
                wb.close()
        return "skip"
    except Exception as exc:  # 실험 도구: 어떤 실패든 기록만 한다
        return f"fail {type(exc).__name__}: {exc}"


def check(path: Path) -> dict:
    with open(path, "rb") as f:
        head = f.read(16)
    return {
        "path": str(path),
        "size": path.stat().st_size,
        "sniff": sniff(head),
        "open": try_open(path),
    }


def collect(args: list[str]) -> list[Path]:
    targets: list[Path] = []
    for a in args:
        p = Path(a)
        if p.is_dir():
            targets += sorted(x for x in p.rglob("*") if x.is_file())
        else:
            targets.append(p)
    return targets


def main(argv: list[str]) -> int:
    rows = [check(p) for p in collect(argv)]
    print(json.dumps(rows, ensure_ascii=False, indent=2))
    return 1 if any(r["sniff"] == "drm" for r in rows) else 0


if __name__ == "__main__":
    sys.exit(main(sys.argv[1:]))
```

```python
# poc/move_check.py
"""실험 2: 공유 폴더 안에서 옮기거나(rename) 복사한 뒤에도 평문이 유지되는지 확인한다.

사용법: python move_check.py <999_LogBook 루트>
전제: <루트>\\_poc\\samples\\ 에 사람이 탐색기로 표본을 복사해 두었다.
"""
import json
import os
import shutil
import sys
from pathlib import Path

from drm_check import check


def run(share_root: Path) -> dict:
    poc = Path(share_root) / "_poc"
    samples = poc / "samples"
    work = poc / "move"
    shutil.rmtree(work, ignore_errors=True)
    staged = work / "staged"
    staged.mkdir(parents=True)

    # ① Python 이 공유 폴더 → 공유 폴더로 복사 (Vault 로 옮기지 않고 사본을 만드는 경우)
    copied = [Path(shutil.copy2(p, staged / p.name)) for p in sorted(samples.iterdir()) if p.is_file()]
    after_copy = [check(p) for p in copied]

    # ② 같은 공유 폴더 안에서 폴더 이름 바꾸기 (Inbox → Vault 이동 방식)
    moved = work / "moved"
    os.rename(staged, moved)
    after_rename = [check(moved / p.name) for p in copied]

    return {"before": [check(p) for p in sorted(samples.iterdir()) if p.is_file()],
            "after_copy": after_copy,
            "after_rename": after_rename}


if __name__ == "__main__":
    print(json.dumps(run(Path(sys.argv[1])), ensure_ascii=False, indent=2))
```

```python
# poc/owner_check.py
"""실험 4: 공유 폴더 파일의 소유자(Owner) 계정을 읽을 수 있는지 확인한다.

사용법: python owner_check.py <폴더>
출력: 파일별 {path, owner, account} JSON 배열. account 가 사번과 대조할 값이다.
"""
import json
import sys
from pathlib import Path

import win32security


def owner_of(path: str) -> str:
    sd = win32security.GetFileSecurity(path, win32security.OWNER_SECURITY_INFORMATION)
    sid = sd.GetSecurityDescriptorOwner()
    name, domain, _ = win32security.LookupAccountSid(None, sid)
    return f"{domain}\\{name}"


def account_name(owner: str) -> str:
    """'HDRND\\A476854' → 'a476854'. 사번 대조는 소문자 계정명으로 한다."""
    return owner.rsplit("\\", 1)[-1].lower()


def main(folder: str) -> list[dict]:
    rows = []
    for p in sorted(Path(folder).rglob("*")):
        if not p.is_file():
            continue
        try:
            owner = owner_of(str(p))
            rows.append({"path": str(p), "owner": owner, "account": account_name(owner)})
        except Exception as exc:
            rows.append({"path": str(p), "owner": None, "error": f"{type(exc).__name__}: {exc}"})
    return rows


if __name__ == "__main__":
    print(json.dumps(main(sys.argv[1]), ensure_ascii=False, indent=2))
```

```python
# poc/bdf_bench.py
"""실험 5: 팀 BDF 를 pyNastran 으로 읽는 시간·메모리·카드 통계를 잰다.

사용법: python bdf_bench.py <BDF 파일 또는 폴더>
폴더를 주면 .bdf .dat .nas 를 모두 측정한다(.blk 는 INCLUDE 조각으로 보고 제외).
"""
import json
import sys
import time
from collections import Counter
from pathlib import Path

import psutil
from pyNastran.bdf.bdf import read_bdf

MODEL_EXTS = {".bdf", ".dat", ".nas"}


def _read(path: Path):
    # 실행 제어부(SOL/CEND)가 없는 조각 파일은 punch=True 로 다시 읽는다
    try:
        return read_bdf(str(path), xref=True, validate=False, punch=False, debug=None)
    except Exception:
        return read_bdf(str(path), xref=True, validate=False, punch=True, debug=None)


def bench(path: Path) -> dict:
    proc = psutil.Process()
    rss0 = proc.memory_info().rss
    t0 = time.perf_counter()
    try:
        model = _read(path)
    except Exception as exc:
        return {"file": path.name, "ok": False, "error": f"{type(exc).__name__}: {exc}"}
    sec = time.perf_counter() - t0
    rss1 = proc.memory_info().rss

    counts = Counter(e.type for e in model.elements.values())
    counts.update(e.type for e in model.rigid_elements.values())
    counts.update(m.type for m in model.masses.values())
    return {
        "file": path.name,
        "ok": True,
        "size_mb": round(path.stat().st_size / 1e6, 1),
        "sec": round(sec, 2),
        "rss_delta_mb": round((rss1 - rss0) / 1e6),
        "nodes": len(model.nodes),
        "counts": dict(counts),
        "rejected": dict(getattr(model, "reject_count", {}) or {}),
        "coords": len(model.coords),
    }


def main(target: str) -> list[dict]:
    p = Path(target)
```

```python
"""실험 7: 260자를 넘는 경로와 한글 파일명을 쓰고 읽고 지울 수 있는지 확인한다.

사용법: python longpath_check.py <999_LogBook 루트>
접두사 없이/있이 두 경우를 모두 시도해 결과를 비교한다.

import json
import os
import shutil
import sys
from pathlib import Path

SEGMENT = "구조해석_검토_자료_폴더_Mooring_Fitting_Fore_Deck"  # 한글 + 영문 긴 폴더명


def to_long(p: str) -> str:
    """Windows 긴 경로 접두사를 붙인다."""
    if p.startswith("\\\\?\\"):
        return p
    if p.startswith("\\\\"):
        return "\\\\?\\UNC\\" + p[2:]
    return "\\\\?\\" + p


def _attempt(base: str) -> str:
    try:
        os.makedirs(base, exist_ok=True)
        f = os.path.join(base, "검토보고서_최종본_리비전3.txt")
        with open(f, "w", encoding="utf-8") as fh:
            fh.write("테스트")
        with open(f, encoding="utf-8") as fh:
            assert fh.read() == "테스트"
        return "ok"
    except Exception as exc:
        return f"fail {type(exc).__name__}: {exc}"


def run(root: Path) -> dict:
    top = str(Path(root) / "_poc" / "longpath")
    deep = os.path.join(top, *([SEGMENT] * 6))
    result = {
        "path_len": len(deep),
        "without_prefix": _attempt(deep),
        "with_prefix": _attempt(to_long(deep)),
    }
    shutil.rmtree(to_long(top), ignore_errors=True)
    return result


if __name__ == "__main__":
    print(json.dumps(run(Path(sys.argv[1])), ensure_ascii=False, indent=2))
```

```python
"""실험 3·6: 크롬 업로드가 평문으로 오는지, 팀원 PC 에서 9095 로 접속되는지 확인한다.

145 에서 실행:
    set LOGBOOK_POC_DIR=\\\\storage.hpc.hd.com\\...\\999_LogBook\\_poc\\web
    uvicorn probe_server:app --host 0.0.0.0 --port 9095
팀원 PC 크롬에서 http://10.14.42.145:9095 접속 → DRM 걸린 로컬 파일 업로드.
"""
import os
from pathlib import Path

from fastapi import FastAPI, Request, UploadFile
from fastapi.responses import HTMLResponse

from drm_check import sniff

SAVE_DIR = Path(os.environ.get("LOGBOOK_POC_DIR", "out/web"))
SAVE_DIR.mkdir(parents=True, exist_ok=True)

app = FastAPI()

PAGE = """<!doctype html><meta charset="utf-8"><title>Logbook PoC</title>
<h1>Logbook 업로드 실험</h1>
<p>접속 IP: {ip}</p>
<input type="file" id="f" multiple>
<pre id="out"></pre>
<script>
document.getElementById('f').onchange = async (e) => {{
  const out = document.getElementById('out');
  for (const file of e.target.files) {{
    const fd = new FormData(); fd.append('file', file);
    const r = await fetch('/upload', {{method: 'POST', body: fd}});
    out.textContent += JSON.stringify(await r.json()) + '\\n';
  }}
}};
</script>"""


@app.get("/", response_class=HTMLResponse)
def index(request: Request) -> str:
    return PAGE.format(ip=request.client.host if request.client else "?")


@app.post("/upload")
async def upload(file: UploadFile) -> dict:
    data = await file.read()
    target = SAVE_DIR / Path(file.filename).name
    target.write_bytes(data)
    with open(target, "rb") as fh:
        saved_head = fh.read(16)
    return {
        "name": file.filename,
        "size": len(data),
        "received_sniff": sniff(data[:16]),
        "saved_sniff": sniff(saved_head),
    }
```
